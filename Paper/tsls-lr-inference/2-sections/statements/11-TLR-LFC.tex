\begin{propositionp}{TLR-LFC}[Least-favorable \(\xi\) Inference]\label{prop:tlr-lfc}
Let \(c_{\varrho,1-\alpha}(\xi) \) denote the \((1-\alpha)^{th}\) {quantile} of the distribution of the TLR estimator fixing \((\varrho,\xi)\).
  We then have:
\begin{align*}
    \P\!\left(\textrm{TLR}(\beta_0) \geq \max_{\xi \in \R^+} c_{\hat\varrho,1-\alpha}(\xi)\right) \to \alpha
\end{align*}
hence \(\textrm{TLR}(\beta_0) \geq \max_{\xi \in \R^+} c_{\hat\varrho,1-\alpha}(\xi)\)  is a valid test for the hypothesis \(\beta=\beta_0\).
\end{propositionp}
