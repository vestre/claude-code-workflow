\begin{lemmap}{LEG-CTE}[Legacy tests, Constant Treatment Effects]\label{lem:leg-cte}
Assume there exists a scalar \(\beta\) such that \(\delta_n=\beta \gamma_n\). We then have, uniformly, the following limits.
    \begin{enumerate}
        \item
        \(\mathrm{AR}(\beta_0)\convd \mathfrak{C}^2_{d_Z}\)
        \item
        \(\mathrm{LM}(\beta_0)\convd \mathfrak{C}^2_1\)
        \item
        \(\mathrm{LR}(\beta_0)\convd \frac{1}{2}\left[\mathfrak{C}^2_{d_Z-1}+\mathfrak{C}^2_1 + \sqrt{(\mathfrak{C}^2_{d_Z}+\mathfrak{C}^2_1)^2 + 4 \cdot \mathfrak{C}^2_1 \cdot \hat r} \right]\)
    \end{enumerate}
\end{lemmap}
