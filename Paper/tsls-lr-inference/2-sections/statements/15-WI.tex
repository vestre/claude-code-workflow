\begin{propositionp}{WI}[Weak-instrument Limiting Distribution]\label{prop:wi-weak}
    Under \Cref{ass:weak-exog}, the limiting distribution of the TLR statistic with weak instruments is given by:
    \begin{align}
    \mathrm{TLR}(\beta_0) \convd \sum_{j=1}^{2d_Z}  W_j^2\left(\frac{\lambda^*  \kappa_j}{1+\lambda^* \kappa_j}\right)^2 \qquad \text{s.t.} \qquad \sum_{j=1}^{2d_Z}\kappa_j w_j^2 = 0
\end{align}
where \(W\sim \mathfrak{N}(w,\boldsymbol{\iota}_{2d_Z})\) and \(W_j\) are the components of this vector.
\end{propositionp}
